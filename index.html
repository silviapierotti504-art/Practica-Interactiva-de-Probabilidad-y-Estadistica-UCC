Generación de datos estocásticos según la naturaleza de la variable if (ctx$tipo == "discreto") { datos <- sample(5:35, 12, replace = TRUE) } else { datos <- round(rnorm(12, mean = sample(50:90, 1), sd = sample(6:14, 1)), 2) } # 3. Se han evaluado correctamente tres medidas fundamentales: la media (", media_val, " ", ctx$unidad, "), la mediana (", mediana_val, " ", ctx$unidad, ") y el desvío estándar (", desvio_val, " ", ctx$unidad, "), manteniendo estrictamente las unidades físicas originales del proceso.") } else if (banco_multimedida == 2) { # --- CASO 2: Combinación de Dispersión Relativa, Asimetría y Posición (CV, Asimetría y Mediana) --- enunciado_p <- paste0("Para la optimización operativa en un proyecto de **", ctx$rama, "**, se analizó una muestra sobre ", ctx$desc, ": [", paste(datos, collapse = ", "), "] ", ctx$unidad, ".\n\n", "Considerando el Coeficiente de Variación (CV), el Coeficiente de Asimetría de Pearson y la Mediana, señale la interpretación analítica correcta:") opcion_correcta <- paste0("El CV es del ", cv_val, "% (medida relativa porcentual), la asimetría de Pearson es de ", asimetria_val, " (indicando el sesgo de la distribución) y la mediana se ubica en ", mediana_val, " ", ctx$unidad, ".") d1 <- paste0("El Coeficiente de Variación se expresa en ", ctx$unidad, ", la asimetría es del ", cv_val, "% y la mediana es adimensional.") d2 <- paste0("El CV es de ", cv_val, " (sin porcentaje), la asimetría supera los ", media_val, " ", ctx$unidad, " y la mediana no se puede calcular en muestras de este tamaño.") d3 <- "El Coeficiente de Variación es siempre idéntico al Coeficiente de Asimetría de Pearson en cualquier muestra de ingeniería." d4 <- "La asimetría se mide en unidades cúbicas del proceso y la mediana representa la variabilidad total." fundamento <- paste0("Correcto. El CV (", cv_val, "%) mide la variabilidad relativa adimensional, el coeficiente de asimetría (", asimetria_val, ") evalúa el sesgo combinando media y desvío, y la mediana (", mediana_val, " ", ctx$unidad, ") indica el percentil 50 en las unidades físicas exactas.") } else if (banco_multimedida == 3) { # --- CASO 3: Análisis de Posición no Central y Rango (Cuartiles Q1, Q3 y Rango total con unidades) --- enunciado_p <- paste0("En un análisis de aseguramiento de calidad para **", ctx$rama, "**, se registraron los siguientes valores sobre ", ctx$desc, ": [", paste(datos, collapse = ", "), "] ", ctx$unidad, ".\n\n", "Evaluando de manera conjunta el Rango total de la muestra, el Primer Cuartil (Q1) y el Tercer Cuartil (Q3), identifique la conclusión correcta:") opcion_correcta <- paste0("El rango total es de ", rango_val, " ", ctx$unidad, " (diferencia entre el máximo y mínimo), mientras que Q1 se ubica en ", q1_val, " ", ctx$unidad, " y Q3 en ", q3_val, " ", ctx$unidad, ", acotando el 50% central de la muestra.") d1 <- paste0("El rango total es adimensional, Q1 es de ", q3_val, " ", ctx$unidad, " y Q3 representa el promedio absoluto de la población.") d2 <- paste0("Q1 y Q3 se miden en unidades cuadradas (", ctx$unidad, "²), y el rango total coincide exactamente con el valor de la mediana.") d3 <- "El rango total es siempre menor que la distancia intercuartílica (Q3 - Q1) en cualquier distribución muestral." d4 <- "Los cuartiles no poseen unidades de medida y no guardan relación con la amplitud de la muestra." fundamento <- paste0("Correcto. Estructuramos la pregunta interactiva con alta exigencia interpretativa question( enunciado_p, answer(opcion_correcta, correct = TRUE), answer(d1), answer(d2), answer(d3), answer(d4), allow_retry = TRUE, random_answer_order = TRUE, correct = fundamento, incorrect = "Incorrecto. Recuerde analizar de forma conjunta las tres medidas solicitadas en la consigna, prestando especial atención a sus dimensiones matemáticas y a las unidades físicas reales del contexto de ingeniería." ) ```
